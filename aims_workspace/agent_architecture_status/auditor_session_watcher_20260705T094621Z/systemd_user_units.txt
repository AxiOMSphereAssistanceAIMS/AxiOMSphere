=== aims-auditor-session-watcher.service ===
[Unit]
Description=AIMS Auditor Session Watcher — checks auditor chain availability
After=network-online.target
Wants=network-online.target

[Service]
Type=oneshot
ExecStart=/home/axi_omi_sphere/aims-workspace/ops/scripts/auditor_session_watcher.sh
WorkingDirectory=/home/axi_omi_sphere/aims-workspace
StandardOutput=journal
StandardError=journal
# No interactive login — if auth required, service exits cleanly
SuccessExitStatus=0 1

=== aims-auditor-session-watcher.timer ===
[Unit]
Description=AIMS Auditor Session Watcher timer — run at login and every 10 minutes
Requires=aims-auditor-session-watcher.service

[Timer]
# Run shortly after user session starts (30 seconds delay)
OnStartupSec=30s
# Then every 10 minutes
OnUnitActiveSec=10min
# Persist so it runs at next login if missed while system was off
Persistent=true
Unit=aims-auditor-session-watcher.service

[Install]
WantedBy=timers.target

=== Install commands (run once as user, no sudo required) ===
mkdir -p ~/.config/systemd/user
cp ops/systemd/user/aims-auditor-session-watcher.service ~/.config/systemd/user/
cp ops/systemd/user/aims-auditor-session-watcher.timer ~/.config/systemd/user/
systemctl --user daemon-reload
systemctl --user enable --now aims-auditor-session-watcher.timer
